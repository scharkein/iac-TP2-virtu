# TP3 - genere par 07-firewall.yml. Remplace le cluster.fw du TP2 en le completant
# (admin-net, cluster-nodes et sg-web du TP2 sont conserves).
[OPTIONS]
enable: 1
policy_in: DROP
policy_out: ACCEPT

[IPSET admin-net]
192.168.56.1

[IPSET admin-workstations]
192.168.56.1

[IPSET cluster-nodes]
192.168.56.11
192.168.56.12

[IPSET dmz-servers]
192.168.10.0/24

[IPSET db-servers]
192.168.20.0/24

[RULES]
IN DROP -source +dmz-servers -p tcp -dport 22,8006 -log warning
IN ACCEPT -source +admin-workstations -p tcp -dport 22
IN ACCEPT -source +admin-workstations -p tcp -dport 8006
IN ACCEPT -source +cluster-nodes

[group sg-web]
IN ACCEPT -p tcp -dport 80
IN ACCEPT -p tcp -dport 443

[group sec-grp-web]
IN ACCEPT -p tcp -dport 80,443
OUT ACCEPT -dest +dc/db-servers -p tcp -dport 3306

[group sec-grp-db]
IN ACCEPT -source +dc/dmz-servers -p tcp -dport 3306
IN DROP -log warning
